\section{Research Projects}
\entryListStart
  \project{SpriteFlow: Flow Matching for Pixel-Art Character Generation (BSc thesis)}
    {\href{https://github.com/mradovic38/sprite-flow}{[code]}}
  \desc{Flow matching model with a time-conditioned U-Net generating 128$\times$128 RGBA sprites; ODE and SDE sampling,

  \project{Proximal Policy Optimization from Scratch}
    {\href{https://github.com/mradovic38/ppo-mujoco}{[PyTorch]} \href{https://github.com/mradovic38/cartpole-ppo}{[JAX]}}
  \desc{Reimplemented PPO from the original paper and compared it with A2C and vanilla policy gradient on MuJoCo
    locomotion (HalfCheetah, Hopper, Walker2d, Swimmer; 3 seeds). Rewrote it in JAX and trained on MuJoCo Playground
    (MJX) Cartpole Balance and Swing-Up.}
  \project{Football Analysis}{\href{https://github.com/mradovic38/football-analysis}{[code]}}
  \desc{Open-source video analysis tool with 175+ GitHub stars: YOLO11 detection, ByteTrack tracking, keypoint-based
    homography for player positions, and K-means team assignment from jersey colors.}
\entryListEnd
